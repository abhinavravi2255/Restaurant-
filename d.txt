let restaurants = [];

const line = () => {

const title = (title) => {
  line();
  console.log(`//-- ${title} --//`);
  line();
};

const createResturant = () => {
  let restaurant;
  title("Create Resturant");

  let id = readline.question("Enter id : ");
  let name = readline.question("Enter name : ");
  let location = readline.question("Enter location : ");

  restaurant = restaurants.find((restaurant) => restaurant.id === id);

  if (restaurant) {
    return title("Resturant with this id already exists");
  }

  restaurant = { id: id, name: name, location: location };

  restaurants.push(restaurant);

  title("Resturant Created!");
};

const displayRestaurants = () => {
  title("Display Resturants");

  if (restaurants.length <= 0) return title("No Resturants Found!");

  restaurants.forEach((restaurant) => {
    console.log(`Resturant id : ${restaurant.id}`);
    console.log(`Resturant name : ${restaurant.name}`);
    console.log(`Resturant location : ${restaurant.location}`);

    line();
    line();
  });
};
